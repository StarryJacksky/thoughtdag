// Does a recursive fs.watch on macOS miss writes made the moment it, or another watcher of the same process, starts or stops?
// One watcher on folder A; before each write into A: nothing (control); a watcher started on folder B just before the write
// (closed 300 ms later); a watcher on B closed just after the write (it was started 300 ms before); or A's own watcher closed
// and started again just before the write. A write counts as missed when A's watcher has not named it by the time it is closed
// (at least 300 ms later, and 0.8 s after the last write); a write reported later than that counts as missed too.
const fs = require('node:fs'); const path = require('node:path'); const os = require('node:os');
const base = fs.realpathSync(fs.mkdtempSync(path.join(os.tmpdir(), 'watch-probe-')));
const A = path.join(base, 'A'), B = path.join(base, 'B'); fs.mkdirSync(A); fs.mkdirSync(B);
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
async function trial(mode, n) {
  const seen = new Set();
  const watchA = () => fs.watch(A, { recursive: true }, (_e, f) => seen.add(String(f)));
  let wa = watchA();
  await sleep(300);
  for (let i = 0; i < n; i++) {
    const name = `${mode}-${i}.txt`;
    const write = () => fs.writeFileSync(path.join(A, name), 'x');
    const watchB = () => fs.watch(B, { recursive: true }, () => {});
    if (mode === 'another starts') { const wb = watchB(); write(); await sleep(300); wb.close(); }
    else if (mode === 'another stops') { const wb = watchB(); await sleep(300); write(); wb.close(); }
    else if (mode === 'it restarts') { wa.close(); wa = watchA(); write(); }
    else write();
    await sleep(300); // long enough for the news to arrive before anything is closed again
  }
  await sleep(500); wa.close();
  const missed = []; for (let i = 0; i < n; i++) if (!seen.has(`${mode}-${i}.txt`)) missed.push(i);
  console.log(`${mode.padEnd(15)} missed ${String(missed.length).padStart(2)} of ${n}${missed.length ? `  (writes ${missed.join(', ')})` : ''}`);
}
(async () => {
  console.log(`node ${process.version}, ${process.platform}`);
  for (const mode of ['control', 'another starts', 'another stops', 'it restarts']) await trial(mode, 60);
  fs.rmSync(base, { recursive: true, force: true });
})();
